cmake_minimum_required(VERSION 3.20)
project(ScrollWheelFix VERSION 0.2.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    add_compile_options(/W4 /utf-8)
else()
    add_compile_options(-Wall -Wextra -Wpedantic)
endif()

add_library(scroll_filter INTERFACE)
target_include_directories(scroll_filter INTERFACE native/common)

include(CTest)
if(BUILD_TESTING)
    add_executable(filter-tests native/tests/filter_tests.cpp)
    target_link_libraries(filter-tests PRIVATE scroll_filter)
    add_test(NAME direction-and-buffer-regressions COMMAND filter-tests)
endif()

if(WIN32)
    add_executable(ScrollWheelFix WIN32 native/windows/main.cpp)
    target_compile_definitions(ScrollWheelFix PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(ScrollWheelFix PRIVATE scroll_filter user32 shell32)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(EVDEV REQUIRED IMPORTED_TARGET libevdev)
    add_executable(scroll-wheel-fix native/linux/main.cpp)
    target_link_libraries(scroll-wheel-fix PRIVATE scroll_filter PkgConfig::EVDEV)
    if(BUILD_TESTING)
        add_executable(linux-frame-tests native/tests/linux_frame_tests.cpp)
        target_include_directories(linux-frame-tests PRIVATE native/linux)
        target_link_libraries(linux-frame-tests PRIVATE scroll_filter)
        add_test(NAME linux-wheel-frames COMMAND linux-frame-tests)
    endif()
endif()
